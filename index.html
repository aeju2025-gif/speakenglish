window.renderStudyCard = function() { if(filteredSentences.length === 0) return; const currentItem = filteredSentences[currentIndex]; const sentenceEl = document.getElementById('study-sentence'); const meaningEl = document.getElementById('study-meaning'); document.getElementById('page-info').innerText = `${currentIndex + 1} / ${filteredSentences.length}`; let rawEng = currentItem.eng; const targetExp = currentItem.expression; const escapeRegExp = (str) => str.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'); // 💡 [개선된 부분] 3인칭 단수, 과거형, 불규칙 동사 등을 똑똑하게 인식하는 정규식 만들기 let words = targetExp.trim().split(/\s+/); let firstWord = escapeRegExp(words[0].toLowerCase()); // 첫 번째 단어가 동사일 확률이 높으므로 변형 형태를 허용합니다. let verbRegex = firstWord; if (firstWord === 'have') verbRegex = '(have|has|had|having)'; else if (firstWord === 'do') verbRegex = '(do|does|did|doing)'; else if (firstWord === 'be') verbRegex = '(be|am|is|are|was|were|been|being)'; else if (firstWord === 'go') verbRegex = '(go|goes|went|gone|going)'; else verbRegex = `${firstWord}(s|es|d|ed|ing)?`; // 일반 동사의 변형 허용 (want -> wants, wanted 등) words[0] = verbRegex; let restOfWords = words.slice(1).map(escapeRegExp).join('\\s+'); // 최종 정규식: (첫단어변형) + (나머지단어) let regexStr = restOfWords ? `\\b${verbRegex}\\s+${restOfWords}\\b` : `\\b${verbRegex}\\b`; const expRegex = new RegExp(regexStr, 'gi'); // 한 문장 안에 핵심 표현이 여러 번 나올 경우를 대비해 매칭된 원래 텍스트를 배열로 저장 let matchTexts = []; rawEng = rawEng.replace(expRegex, (match) => { matchTexts.push(match); return "___EXPRESSION_PLACEHOLDER___"; }); const tokens = rawEng.split(/(\s+)/); let processedHTML = ""; tokens.forEach(token => { if (token.trim() === "") processedHTML += token; else if (token === "___EXPRESSION_PLACEHOLDER___") { // 저장해둔 실제 매칭 단어(예: wants to, has to)를 그대로 하이라이트 처리 processedHTML += `${matchTexts.shift()}`; } else { const match = token.match(/^([a-zA-Z0-9'-]+)([^a-zA-Z0-9'-]*)$/); if (match) { processedHTML += `${match[1]}${match[2]}`; } else processedHTML += token; } }); sentenceEl.innerHTML = processedHTML; meaningEl.innerText = currentItem.kor; document.getElementById('next-btn').innerText = (currentIndex === filteredSentences.length - 1) ? "학습 완료" : "다음 문장 ➡️"; }
